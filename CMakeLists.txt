cmake_minimum_required(VERSION 3.20)
project(elan VERSION 3.6 LANGUAGES C CXX)

# Strict C17/C++17, warnings are errors (S2). The *_COMPAT_OPTIONS names are
# historical (S1 built the sources in pre-standard modes).
# -fsigned-char: the code stores -1 in `char` (e.g. module visibilities); char
# is unsigned on Linux/aarch64, where local rules then become visible.
set(ELAN_C_COMPAT_OPTIONS   -std=c17 -fsigned-char -Wall -Wextra -Werror)
set(ELAN_CXX_COMPAT_OPTIONS -std=c++17 -fsigned-char -Wall -Wextra -Werror)
set(ELAN_DEFINITIONS
    PACKAGE="elan-interpreter" VERSION="3.6g" YYTEXT_POINTER=1)

# make check-sanitize: sanitizer build (ELAN_SANITIZERS, default address,undefined)
option(ELAN_SANITIZE "Build with sanitizers" OFF)
set(ELAN_SANITIZERS "address,undefined" CACHE STRING "-fsanitize= list")

# The compiler (elanc, REM, runtime libraries: spec S5a). In a sanitizer
# build, its libraries and the programs generated by elanc are built with
# UBSan only (spec S5b): libchoice saves and restores the C stack by copying
# it (cpl), which AddressSanitizer reports by design (its redzones are in the
# copied stack).
option(ELAN_COMPILER "Build the compiler (needs a JDK, bison, flex, bdw-gc)" ON)
set(ELAN_COMPILER_SANITIZE_OPTIONS "")
if(ELAN_SANITIZE AND ",${ELAN_SANITIZERS}," MATCHES ",undefined,")
  set(ELAN_COMPILER_SANITIZE_OPTIONS -fsanitize=undefined -fno-sanitize-recover=undefined
                                     -fno-omit-frame-pointer -g)
endif()
if(ELAN_COMPILER)
  add_subdirectory(src/compiler)
endif()

# The compiler in Rust (spec S6): optional since S7e (spec G7: elanc-go has
# reached parity), built with -DELAN_RUST=ON when cargo is found.
option(ELAN_RUST "Build the Rust compiler elanc-rs (needs cargo)" OFF)
if(ELAN_RUST)
  find_program(ELAN_CARGO cargo)
  if(ELAN_CARGO)
    add_subdirectory(src/compiler-rs)
  else()
    message(STATUS "cargo not found: elanc-rs is not built")
  endif()
endif()

# The compiler in Go (spec S7): only when go >= 1.24 is found.
option(ELAN_GO "Build the Go compiler elanc-go (needs go >= 1.24)" ON)
if(ELAN_GO)
  find_program(ELAN_GO_EXE go)
  if(ELAN_GO_EXE)
    # the local toolchain: the one that builds elanc-go (GOTOOLCHAIN=local),
    # not one go would download or switch to
    execute_process(COMMAND ${CMAKE_COMMAND} -E env GOTOOLCHAIN=local ${ELAN_GO_EXE} env GOVERSION
                    OUTPUT_VARIABLE ELAN_GO_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
    string(REGEX MATCH "go([0-9]+)\\.([0-9]+)" _m "${ELAN_GO_VERSION}")
    if(CMAKE_MATCH_1 GREATER 1 OR (CMAKE_MATCH_1 EQUAL 1 AND CMAKE_MATCH_2 GREATER_EQUAL 24))
      add_subdirectory(src/compiler-go)
      set(ELAN_GO_BUILT ON)
    else()
      message(STATUS "${ELAN_GO_VERSION} is older than go1.24: elanc-go is not built")
    endif()
  else()
    message(STATUS "go not found: elanc-go is not built")
  endif()
endif()

# What bin/elanc is (S7e, decision of 2026-10-11): a relative symlink to
# elanc-go (go, the default) or to elanc-rem, REM's driver (rem: the install
# before S7e). go falls back to rem when elanc-go is not built (no go >=
# 1.24, -DELAN_GO=OFF), so that a machine without Go still gets an elanc.
set(ELAN_ELANC go CACHE STRING "what bin/elanc is: go (elanc-go) or rem (elanc-rem)")
set_property(CACHE ELAN_ELANC PROPERTY STRINGS go rem)
if(NOT ELAN_ELANC MATCHES "^(go|rem)$")
  message(FATAL_ERROR "ELAN_ELANC must be go or rem, not '${ELAN_ELANC}'")
endif()
set(ELAN_ELANC_TARGET "")
if(ELAN_ELANC STREQUAL "rem")
  if(NOT ELAN_COMPILER)
    message(FATAL_ERROR "ELAN_ELANC=rem needs the REM compiler (-DELAN_COMPILER=ON)")
  endif()
  set(ELAN_ELANC_TARGET elanc-rem)
  message(STATUS "bin/elanc is elanc-rem (ELAN_ELANC=rem)")
elseif(ELAN_GO_BUILT)
  set(ELAN_ELANC_TARGET elanc-go)
  message(STATUS "bin/elanc is elanc-go (ELAN_ELANC=go)")
elseif(ELAN_COMPILER)
  set(ELAN_ELANC_TARGET elanc-rem)
  message(STATUS "ELAN_ELANC=go but elanc-go is not built: bin/elanc is elanc-rem")
else()
  message(STATUS "no compiler is built: bin/elanc is not installed")
endif()
# (the link is made in the build directory and installed as a file: it is
# then in install_manifest.txt, which an uninstall reads)
set(_elanc_link ${CMAKE_BINARY_DIR}/elanc-link/elanc)
file(REMOVE ${_elanc_link})
if(ELAN_ELANC_TARGET)
  file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/elanc-link)
  file(CREATE_LINK ${ELAN_ELANC_TARGET} ${_elanc_link} SYMBOLIC)
  install(FILES ${_elanc_link} DESTINATION bin)
endif()

if(ELAN_SANITIZE)
  add_compile_options(-fsanitize=${ELAN_SANITIZERS} -fno-sanitize-recover=undefined
                      -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${ELAN_SANITIZERS})
endif()

enable_testing()   # unit tests are declared in src/interpreter
add_subdirectory(src/interpreter)
